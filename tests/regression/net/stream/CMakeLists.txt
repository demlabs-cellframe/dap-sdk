cmake_minimum_required(VERSION 3.16)
project(stream_keepalive_regression C)
enable_testing()
get_filename_component(SDK_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
add_executable(test_stream_keepalive test_stream_keepalive.c)
file(GLOB_RECURSE SDK_HEADERS "${SDK_ROOT}/net/*.h" "${SDK_ROOT}/core/*.h"
    "${SDK_ROOT}/crypto/include/*.h" "${SDK_ROOT}/io/include/*.h")
foreach(HEADER IN LISTS SDK_HEADERS)
    get_filename_component(HEADER_DIR "${HEADER}" DIRECTORY)
    target_include_directories(test_stream_keepalive PRIVATE "${HEADER_DIR}")
endforeach()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
    add_subdirectory("${SDK_ROOT}/3rdparty/json-c" json-c EXCLUDE_FROM_ALL)
endif()
target_include_directories(test_stream_keepalive PRIVATE $<TARGET_PROPERTY:dap_json-c,INTERFACE_INCLUDE_DIRECTORIES>)
target_include_directories(test_stream_keepalive PRIVATE "${SDK_ROOT}/3rdparty/uthash/src"
    "${SDK_ROOT}/3rdparty"
    "${SDK_ROOT}/3rdparty/json-c")
target_compile_definitions(test_stream_keepalive PRIVATE DAP_OS_LINUX _GNU_SOURCE)
target_compile_options(test_stream_keepalive PRIVATE -O2 -UNDEBUG -ffunction-sections -fdata-sections)
target_link_options(test_stream_keepalive PRIVATE -Wl,--gc-sections)
target_link_libraries(test_stream_keepalive PRIVATE pthread)
add_test(NAME test_stream_keepalive COMMAND test_stream_keepalive)
set_tests_properties(test_stream_keepalive PROPERTIES TIMEOUT 10 LABELS "regression;stream;keepalive")
